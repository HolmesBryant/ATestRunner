ATestRunner

A modern, flexible JavaScript test runner for the browser.

Demo (console) | Demo (DOM)

README

ATestRunner is a comprehensive suite for defining, running, and reporting tests which require a DOM environment. It operates on a queue-based system, allowing for asynchronous test execution with flexible output options to the console or a specified DOM element.

Installation

Download `ATestRunner.min.js` from the `src` directory and include it in an HTML page.

Quick Start

The default behavior is to log test results to the console.

Sample Test Suite

/******************** * The test suite * /tests/MyApp.tests.js *********************/ import app from '../src/MyApp.js'; import ATestRunner from './ATestrunner.js'; const runner = new ATestRunner(import.meta.url) const { benchmark, equal, genCombos, group, info, log, skip, spyOn, test, throws, wait, when } = runner; // Helper functions function setup() { // set up app } function teardown() { // tear down app } // A simple message info("Testing MyApp"); // Group tests using group() group("Testing Basic Properties", () => { // setup() will run before any tests in the group setup(); // You don't have to wrap simple assertions in a function test( "app.foo should be 'foo'", app.foo, 'foo' ); // However, you can if you want. test( 'bar should be null', () => app.bar, null ); /* If an expression might throw an error, you should wrap it in a function. */ test( "This is reported as an error", () => app.mightThrowError(true), null ); /* You can also use throws() to check if a function throws an error. throws(testFn, ...args) */ test( "This test passes", throws( app.mightThrowError, 'badArg' ), true ); // This test is skipped, but still logged skip( "arr should be an Array", Array.isArray(app.arr), true ); // teardown() will run after all tests in the group teardown(); }); try { test( "This is also reported as an error", non.existant.func(), 'expected value' ); } catch (error) { runner.handleError(error) } test( "Use wait(milliseconds) if you need to wait a few ticks", async () => { const div = document.createElement('div'); documment.body.append(div); // wait a few ticks await wait(10); const result = document.body.querySelector('div') !== null; div.remove(); return result; }, true ); /* When testing async operations, wrap the assertion in an async function */ test( "asyncFunc() should return 'foo'", async () => await app.asyncFunc(), 'foo' ); // when() is useful when testing async or DOM operations. // async when(expression, timeoutMs = 1000, checkIntervalMs = 100) /* If you just need the return value from an async operation, you don't have to await it (but you can) */ test( "asyncFunc should return 'foo'", when( app.asyncFunc('foo') ), 'foo' ); /* if you need to compare the return value with something else, you *must* use await */ test( "asyncFunc === 'foo'", when( await app.asyncFunc('foo') === 'foo' ), true ); // equal() performs a deep equality comparison. test( "getPojo() should return {a:1, b:2, c:3}", equal(app.getPojo(), {a:1, b:2, c:3}), true ); group("Use genCombos() for dynamically created tests", () => { /** * *genCombos(options = {}) * First iteration yields { a: 1, b: 'c' } * Second iteration yields { a: 2, b: 'c' } */ const options = { a: [1, 2], b: 'c' }; for (const combo of genCombos(options)) { // Dynamically create a test for each combination test( `Combination with a=${combo.a} should have b='c'`, combo.b, 'c' ); // log each combo using log(label, data) log('snapshot', combo); } }); // Use benchmark() to evaluate a function's performance // async benchmark(fn, nubmerOfTimesToRun = 1, thisArg = null, ...args) test( "asyncFunc() should complete in under 1 second", async () => await benchmark(app.asyncFunc, 1, null, 'arg1', 'arg2') < 1000, true ); group("Create a spy to observe an object or function", () => { // spyOn(object, methodName) const spy = spyOn(console, 'debug'); console.debug('called when testing spyOn()', 'foo'); test( "console.debug was called 1 time", spy.callCount, 1 ); test( "console.debug was called with arg 'foo'", spy.calls[0][1], 'foo' ); spy.restore(); }); // Run the tests after they have all been defined. runner.run();
